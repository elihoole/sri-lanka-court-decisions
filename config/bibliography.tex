% Bibliography Formatting Customization
% Overrides OSCOLA defaults for better readability

% Float placement - pin figures and tables to top of page
\usepackage{float}
\floatplacement{figure}{t}
\floatplacement{table}{t}

% Reduce spacing around figures
\setlength{\intextsep}{10pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}

% Style Pandoc citeproc bibliography entries
% Add vertical spacing between references (0.8em)
\usepackage{etoolbox}
\newlength{\cslentrygap}
\setlength{\cslentrygap}{0.8em plus 0.2em minus 0.1em}
\ifdefined\cslentryspacingunit
  % pandoc <= 3.1.6: CSLReferences spaces entries as paragraphs; override
  % its entry-spacing unit so the template's own \parskip setting uses our gap
  \setlength{\cslentryspacingunit}{\cslentrygap}
\else
  % pandoc >= 3.1.7: CSLReferences is a list; set the gap via \itemsep
  \renewenvironment{CSLReferences}[2]
   {\begin{list}{}{%
    \setlength{\itemindent}{0pt}%
    \setlength{\leftmargin}{0pt}%
    \setlength{\parsep}{0pt}%
    \ifodd #1
     \setlength{\leftmargin}{\cslhangindent}%
     \setlength{\itemindent}{-1\cslhangindent}%
    \fi
    \setlength{\itemsep}{\cslentrygap}}}
   {\end{list}}
\fi

% Remove angle brackets from URLs
\usepackage{url}
\urlstyle{same}
\makeatletter
\def\url@leostyle{%
  \@ifundefined{selectfont}{\def\UrlFont{\sf}}{\def\UrlFont{\small\ttfamily}}}
\makeatother

% Increase vertical spacing between footnotes
\setlength{\footnotesep}{1.2\baselineskip}

% Add space between footnote number and text
\usepackage[hang]{footmisc}
\setlength{\footnotemargin}{1em}
