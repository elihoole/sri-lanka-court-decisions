% Bibliography Formatting Customization
% Overrides OSCOLA defaults for better readability

% Figure placement - prevent images from floating to separate pages
\usepackage{float}
\let\origfigure\figure
\let\endorigfigure\endfigure
\renewenvironment{figure}[1][2] {
    \expandafter\origfigure\expandafter[H]
} {
    \endorigfigure
}

% Reduce spacing around figures
\setlength{\intextsep}{10pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}

% Style Pandoc citeproc bibliography entries
% Add vertical spacing between references
\usepackage{etoolbox}
\AtBeginEnvironment{CSLReferences}{%
  \setlength{\parskip}{0.8em plus 0.2em minus 0.1em}%
}

% Remove angle brackets from URLs
\usepackage{url}
\urlstyle{same}
\makeatletter
\def\url@leostyle{%
  \@ifundefined{selectfont}{\def\UrlFont{\sf}}{\def\UrlFont{\small\ttfamily}}}
\makeatother

% Increase vertical spacing between footnotes
\setlength{\footnotesep}{1.2\baselineskip}

% Add space between footnote number and text
\usepackage[hang]{footmisc}
\setlength{\footnotemargin}{1em}
